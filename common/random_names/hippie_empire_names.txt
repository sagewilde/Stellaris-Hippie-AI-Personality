# Naturalist Commune 1
empire_name_format = {
	random_weight = {
		factor = 0
		modifier = {
			add = 1
			has_government = "gov_naturalist_commune"
			is_pirate = no
			is_primitive = no
			NOT = { is_country_type = fallen_empire }
			NOT = { is_country_type = awakened_fallen_empire }
		}
	}
	format = "{AofB {<hippie_nat> [This.GetCapitalSystemNameOrRandom]}}"
	noun = "{[This.GetCapitalSystemNameOrRandom]}"
	prefix_format = "{AofBpfx{[This.GetCapitalSystemNameOrRandom] <hippie_nat>}}"
	# Empire of Sol
}
